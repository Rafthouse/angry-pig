cmake_minimum_required(VERSION 3.24)
project(AngryPigSpike LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(IPLUG_DEPLOY_PLUGINS OFF CACHE BOOL "No production install" FORCE)
set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../deps/iplug2" CACHE PATH "Pinned iPlug2")
include(${IPLUG2_DIR}/iPlug2.cmake)
find_package(iPlug2 REQUIRED)
file(GLOB_RECURSE AP_WEB CONFIGURE_DEPENDS resources/web/*)
if(APPLE)
 set(AP_FORMATS VST3 AU)
else()
 set(AP_FORMATS VST3)
endif()
option(AP_DIAGNOSTICS "Enable lab trace/probes/hooks" OFF)
option(AP_MATCH_DSP "Use the isolated measurement-derived Angry Pig match engine" ON)
if(AP_DIAGNOSTICS)
 set(AP_EXTRA APProbe.cpp)
endif()
iplug_add_plugin(AngryPigSpike SOURCES ${AP_EXTRA} AngryPigSpike.cpp AngryPigSpike.h resources/resource.h WEB_RESOURCES ${AP_WEB} UI WEBVIEW FORMATS ${AP_FORMATS})
if(MSVC)
 target_compile_options(AngryPigSpike-vst3 PRIVATE /fp:strict)
endif()

if(WIN32)
add_custom_command(TARGET AngryPigSpike-vst3 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/resources/web ${CMAKE_BINARY_DIR}/out/AngryPigSpike.vst3/Contents/Resources/web)
endif()

target_compile_definitions(AngryPigSpike-vst3 PRIVATE AP_AUTHORITATIVE_BYPASS_ID=3 AP_NATIVE_ASPECT_RESIZE=1)
if(MSVC AND AP_DIAGNOSTICS)
set(AP_PROBED_SOURCES AngryPigSpike.cpp "${IPLUG2_DIR}/IPlug/IPlugAPIBase.cpp" "${IPLUG2_DIR}/IPlug/IPlugProcessor.cpp" "${IPLUG2_DIR}/IPlug/VST3/IPlugVST3.cpp" "${IPLUG2_DIR}/IPlug/VST3/IPlugVST3_ProcessorBase.cpp")
set_source_files_properties(${AP_PROBED_SOURCES} PROPERTIES COMPILE_OPTIONS "/FI${CMAKE_CURRENT_SOURCE_DIR}/APProbe.h")
endif()
if(WIN32 AND MSVC AND NOT AP_MATCH_DSP)
if(AP_DIAGNOSTICS)
 set(AP_C_PROBE ../dsp/probe.c)
endif()
add_library(ap-c-dsp STATIC ../qualification/safe_pig.c ../dsp/pig.c ${AP_C_PROBE} ../dsp/rt/wasm-rt-impl.c ../dsp/rt/wasm-rt-mem-impl.c ../dsp/rt/wasm-rt-exceptions-impl.c)
target_include_directories(ap-c-dsp PUBLIC ../qualification ../dsp ../dsp/rt ../deps/wabt-1.0.39/include)
target_compile_definitions(ap-c-dsp PUBLIC WASM_RT_USE_MMAP=0 WASM_RT_MEMCHECK_BOUNDS_CHECK=1 WASM_RT_MEMCHECK_GUARD_PAGES=0)
target_compile_options(ap-c-dsp PRIVATE /O2 /fp:strict)
if(AP_DIAGNOSTICS)
set_source_files_properties(../qualification/safe_pig.c ../dsp/pig.c ../dsp/rt/wasm-rt-impl.c ../dsp/rt/wasm-rt-mem-impl.c PROPERTIES COMPILE_OPTIONS "/FI${CMAKE_CURRENT_SOURCE_DIR}/../dsp/probe.h")
endif()
target_link_libraries(AngryPigSpike-vst3 PRIVATE ap-c-dsp)
endif()

get_filename_component(AP_CACHE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../runtime-cache" ABSOLUTE)
target_compile_definitions(AngryPigSpike-vst3 PRIVATE AP_SPIKE_DEFAULT_CACHE="${AP_CACHE_ROOT}")

target_compile_definitions(AngryPigSpike-vst3 PRIVATE AP_DIAGNOSTICS=$<BOOL:${AP_DIAGNOSTICS}>)
if(TARGET ap-c-dsp)
 target_compile_definitions(ap-c-dsp PRIVATE AP_DIAGNOSTICS=$<BOOL:${AP_DIAGNOSTICS}>)
endif()

option(AP_TIMING "WP1 bounded stage timing" OFF)
if(AP_TIMING AND NOT AP_DIAGNOSTICS)
 message(FATAL_ERROR "WP1 timing requires diagnostic lab exports")
endif()
target_compile_definitions(AngryPigSpike-vst3 PRIVATE AP_TIMING=$<BOOL:${AP_TIMING}>)
if(TARGET ap-c-dsp)
 target_compile_definitions(ap-c-dsp PRIVATE AP_TIMING=$<BOOL:${AP_TIMING}>)
endif()
target_include_directories(AngryPigSpike-vst3 PRIVATE ../qualification)
if(AP_TIMING)
 target_sources(AngryPigSpike-vst3 PRIVATE APTiming.cpp)
endif()

# RC1 diagnostic-only test branch: RT allocation/lock counters (probe.c from the AP-SF source). Production leaves AP_DIAGNOSTICS OFF.
if(AP_DIAGNOSTICS)
 target_sources(AngryPigSpike-vst3 PRIVATE ../dsp/probe.c)
 set_source_files_properties(../dsp/probe.c PROPERTIES COMPILE_DEFINITIONS AP_PROBE_IMPL)
endif()
